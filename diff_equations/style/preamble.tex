% xcolor грузится первым и с опциями: tikz и tcolorbox подтягивают его
% сами, и повторный \usepackage[...]{xcolor} дал бы «Option clash».
\usepackage[dvipsnames]{xcolor}

\input{style/fonts-geometry}   % шрифты, язык, геометрия, интервалы
\input{style/math-commands}    % amsmath, нормы, обычные theorem-окружения
\input{style/tikz-styles}      % tikz и наборы стилей иллюстраций
\input{style/highlight}      % палитра хайлайтов и команды \hlY, \hlB, ...
\input{style/theorem-boxes}    % цветные боксы + пометка required (грузит hyperref)
\input{style/cover}            % обложка, составители, оглавление
